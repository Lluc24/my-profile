% Minimal business card in GitToken's liquid-glass style: a night aurora
% behind one frosted pane per side. Compile with: lualatex card.tex
% TikZ runs at x=1mm,y=1mm, so every bare coordinate is a millimetre.
\documentclass[tikz,border=0mm]{standalone}
\usepackage{graphicx}
\usepackage{fontspec}
\usetikzlibrary{calc,fadings}
\usepackage{hyperref}
\hypersetup{hidelinks}

\setmainfont{Inter}[UprightFont=*-Regular, BoldFont=*-ExtraBold]
\newfontfamily\semiboldfont{Inter SemiBold}
% Programming ligatures off so URLs print character for character.
\setmonofont{JetBrains Mono}[RawFeature={-calt;-liga}]

% Palette from GitToken's frontend: --color-night in src/index.css and the
% dark-theme aurora blobs in src/App.tsx.
\definecolor{night}{HTML}{05060F}
\definecolor{ink}{HTML}{FFFFFF}
\colorlet{secondaryink}{ink!60!night}
\definecolor{auroracyan}{HTML}{06B6D4}
\definecolor{auroraviolet}{HTML}{7C3AED}
\definecolor{aurorafuchsia}{HTML}{D946EF}

\tikzfading[name=aurorablob, inner color=transparent!0, outer color=transparent!100]
\tikzfading[name=glassfill, left color=transparent!85, right color=transparent!95]

\def\cardwidth{85}
\def\cardheight{55}
\def\bleed{3}
\def\glassradius{4}
\def\textleft{10}
\pgfmathsetmacro{\fullwidth}{\cardwidth+2*\bleed}
\pgfmathsetmacro{\fullheight}{\cardheight+2*\bleed}

% The PDF page is the bleed box; the trim box tells the print shop where to cut.
\pdfvariable pageattr{/TrimBox [8.504 8.504 249.447 164.409] /BleedBox [0 0 257.953 172.913]}

% #1, #2, #3: centres of the cyan, violet and fuchsia blobs.
\newcommand{\aurora}[3]{%
  \fill[night] (-\bleed,-\bleed) rectangle (\cardwidth+\bleed,\cardheight+\bleed);
  \fill[auroracyan, opacity=0.55, path fading=aurorablob] #1 circle (40);
  \fill[auroraviolet, opacity=0.55, path fading=aurorablob] #2 circle (38);
  \fill[aurorafuchsia, opacity=0.55, path fading=aurorablob] #3 circle (32);}

% After src/glass.ts: white fill fading from 15% to 5% at 135deg, a 20% white
% border, and a 45% white inset highlight along the top edge and corners.
% #1, #2: lower-left and upper-right corners of the pane.
\newcommand{\glasspane}[2]{%
  \fill[white, path fading=glassfill, fading angle=-45, rounded corners=\glassradius mm] #1 rectangle #2;
  \draw[white, opacity=0.2, line width=0.15mm, rounded corners=\glassradius mm] #1 rectangle #2;
  \begin{scope}
    \path #1 coordinate (paneBottomLeft) #2 coordinate (paneTopRight);
    \clip ($(paneBottomLeft |- paneTopRight)+(0,-\glassradius)$) rectangle (paneTopRight);
    \draw[white, opacity=0.45, line width=0.15mm, rounded corners=\glassradius mm - 0.15mm]
      ($(paneBottomLeft)+(0.15,0.15)$) rectangle ($(paneTopRight)-(0.15,0.15)$);
  \end{scope}}

\newcommand{\linktext}[2]{\href{#1}{\ttfamily\fontsize{6.5}{7}\selectfont\color{ink}#2}}

% #1: baseline y in mm; #2: link target; #3: printed text.
\newcommand{\contactline}[3]{%
  \node[anchor=base west, inner sep=0] at (\textleft,#1) {\linktext{#2}{#3}};}
