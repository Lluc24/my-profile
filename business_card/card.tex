% Business card in the GitToken style: night background, aurora glows,
% Inter type, glass panel, iridescent rim. Compile with: lualatex card.tex
% TikZ runs at x=1mm,y=1mm, so every bare coordinate is a millimetre.
\documentclass[tikz,border=0mm]{standalone}
\usepackage{graphicx}
\usepackage{fontspec}
\usetikzlibrary{fadings}
\usepackage{hyperref}
\hypersetup{hidelinks}

\setmainfont{Inter}[UprightFont=*-Regular, BoldFont=*-ExtraBold]
\newfontfamily\semiboldfont{Inter SemiBold}
\newfontfamily\mediumfont{Inter Medium}
% Programming ligatures off so URLs print character for character.
\setmonofont{JetBrains Mono}[RawFeature={-calt;-liga}]

% Palette from GitToken: frontend/src/index.css, the dark aurora in App.tsx
% and the legendary rarity in backend templates/palette.jinja.
\definecolor{night}{HTML}{0B0E1A}
\definecolor{ink}{HTML}{FFFFFF}
\definecolor{glowcyan}{HTML}{06B6D4}
\definecolor{glowviolet}{HTML}{7C3AED}
\definecolor{glowfuchsia}{HTML}{D946EF}
\definecolor{accent}{HTML}{67E8F9}
\definecolor{lightviolet}{HTML}{A78BFA}
\definecolor{lightfuchsia}{HTML}{E879F9}
\definecolor{lightamber}{HTML}{FCD34D}
\definecolor{lightgreen}{HTML}{86EFAC}
\colorlet{secondaryink}{ink!60!night}

\tikzfading[name=glow, inner color=transparent!0, outer color=transparent!100]
\pgfdeclarehorizontalshading{iridescent}{100bp}{%
  color(0bp)=(accent); color(25bp)=(lightfuchsia); color(50bp)=(lightamber);
  color(75bp)=(lightgreen); color(100bp)=(accent)}
\pgfdeclarehorizontalshading{brandtile}{100bp}{%
  color(0bp)=(accent); color(50bp)=(lightviolet); color(100bp)=(lightfuchsia)}

\def\cardwidth{85}
\def\cardheight{55}
\def\bleed{3}
\pgfmathsetmacro{\fullwidth}{\cardwidth+2*\bleed}
\pgfmathsetmacro{\fullheight}{\cardheight+2*\bleed}

% The PDF page is the bleed box; the trim box tells the print shop where to cut.
\pdfvariable pageattr{/TrimBox [8.504 8.504 249.447 164.409] /BleedBox [0 0 257.953 172.913]}

% #1 and #2: centres of the cyan and fuchsia glows; violet sits on the right edge.
\newcommand{\nightbackground}[2]{%
  \fill[night] (-\bleed,-\bleed) rectangle (\cardwidth+\bleed,\cardheight+\bleed);
  \begin{scope}
    \clip (-\bleed,-\bleed) rectangle (\cardwidth+\bleed,\cardheight+\bleed);
    \fill[glowcyan, opacity=0.5, path fading=glow] #1 circle (38);
    \fill[glowviolet, opacity=0.55, path fading=glow] (\cardwidth+4,30) circle (34);
    \fill[glowfuchsia, opacity=0.45, path fading=glow] #2 circle (30);
  \end{scope}
}

\newcommand{\spacedcaps}[2]{\addfontfeature{LetterSpace=#1}\MakeUppercase{#2}}

% #1: baseline y in mm.
\newcommand{\contactrow}[4]{%
  \node[anchor=base west, inner sep=0] at (9,#1)
    {\mediumfont\fontsize{6.5}{7}\selectfont\color{secondaryink}#2};
  \node[anchor=base east, inner sep=0] at (76,#1)
    {\href{#3}{\ttfamily\fontsize{6.5}{7}\selectfont\color{ink}#4}};}
