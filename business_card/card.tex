% Minimal business card in the GitToken style: night background, Inter type,
% one iridescent accent. Compile with: lualatex card.tex
% TikZ runs at x=1mm,y=1mm, so every bare coordinate is a millimetre.
\documentclass[tikz,border=0mm]{standalone}
\usepackage{graphicx}
\usepackage{fontspec}
\usepackage{hyperref}
\hypersetup{hidelinks}

\setmainfont{Inter}[UprightFont=*-Regular, BoldFont=*-ExtraBold]
\newfontfamily\semiboldfont{Inter SemiBold}
% Programming ligatures off so URLs print character for character.
\setmonofont{JetBrains Mono}[RawFeature={-calt;-liga}]

% Palette from GitToken's backend templates/palette.jinja: the dark card
% background and the iridescent rim of legendary profile cards.
\definecolor{night}{HTML}{0B0E1A}
\definecolor{ink}{HTML}{FFFFFF}
\colorlet{secondaryink}{ink!60!night}
\definecolor{lightcyan}{HTML}{67E8F9}
\definecolor{lightfuchsia}{HTML}{E879F9}
\definecolor{lightamber}{HTML}{FCD34D}
\definecolor{lightgreen}{HTML}{86EFAC}

% PGF maps a shaded path onto the middle 50bp of a 100bp shading.
\pgfdeclarehorizontalshading{iridescent}{100bp}{%
  color(0bp)=(lightcyan); color(25bp)=(lightcyan); color(42bp)=(lightfuchsia);
  color(58bp)=(lightamber); color(75bp)=(lightgreen); color(100bp)=(lightgreen)}

\def\cardwidth{85}
\def\cardheight{55}
\def\bleed{3}
\pgfmathsetmacro{\fullwidth}{\cardwidth+2*\bleed}
\pgfmathsetmacro{\fullheight}{\cardheight+2*\bleed}

% The PDF page is the bleed box; the trim box tells the print shop where to cut.
\pdfvariable pageattr{/TrimBox [8.504 8.504 249.447 164.409] /BleedBox [0 0 257.953 172.913]}

\def\leftmargin{8}

\newcommand{\nightbackground}{%
  \fill[night] (-\bleed,-\bleed) rectangle (\cardwidth+\bleed,\cardheight+\bleed);}

\newcommand{\linktext}[2]{\href{#1}{\ttfamily\fontsize{6.5}{7}\selectfont\color{ink}#2}}

% #1: left end of the line, in mm.
\newcommand{\accentline}[1]{\shade[shading=iridescent] #1 rectangle ++(12,0.5);}

% #1: baseline y in mm; #2: link target; #3: printed text.
\newcommand{\contactline}[3]{%
  \node[anchor=base west, inner sep=0] at (\leftmargin,#1) {\linktext{#2}{#3}};}

\begin{document}

\begin{tikzpicture}[x=1mm,y=1mm]
  \useasboundingbox (0,0) rectangle (\fullwidth,\fullheight);
  \begin{scope}[shift={(\bleed,\bleed)}]
    \nightbackground

    \node[anchor=base west, inner sep=0] at (\leftmargin,41)
      {\fontsize{13}{14}\selectfont\bfseries\addfontfeature{LetterSpace=-2}\color{ink}Lluc Santamaria Riba};
    \node[anchor=base west, inner sep=0] at (\leftmargin,35.5)
      {\semiboldfont\fontsize{6}{7}\selectfont\addfontfeature{LetterSpace=30}\color{secondaryink}\MakeUppercase{Voice AI Engineer · Prosper AI}};
    \accentline{(\leftmargin,30.5)}

    \contactline{19}{https://lluc.sh}{lluc.sh}
    \contactline{15}{mailto:lluc@lluc.sh}{lluc@lluc.sh}
    \contactline{11}{https://github.com/Lluc24}{github.com/Lluc24}
    \contactline{7}{https://linkedin.com/in/lluc-santamaria}{linkedin.com/in/lluc-santamaria}
  \end{scope}
\end{tikzpicture}

\begin{tikzpicture}[x=1mm,y=1mm]
  \useasboundingbox (0,0) rectangle (\fullwidth,\fullheight);
  \begin{scope}[shift={(\bleed,\bleed)}]
    \nightbackground

    % The QR quiet zone must stay white whatever ink becomes.
    \fill[white, rounded corners=1.5mm] (30,18) rectangle (55,43);
    \node[anchor=center, inner sep=0] at (42.5,30.5)
      {\href{https://lluc.sh}{\includegraphics[width=21mm]{qr.png}}};
    \accentline{(36.5,12.5)}
    \node[anchor=base, inner sep=0] at (42.5,7)
      {\linktext{https://lluc.sh}{lluc.sh}};
  \end{scope}
\end{tikzpicture}

\end{document}
